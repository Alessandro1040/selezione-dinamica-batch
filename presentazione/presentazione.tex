\begin{frame}
  \begin{center}
    {\LARGE\bfseries\color{acc} Selezione Dinamica della Dimensione del Campione}\\[0.3cm]
    {\large in Metodi di Ottimizzazione per il Machine Learning}\\[0.6cm]
    \textcolor{acc2}{\rule{0.7\textwidth}{0.8pt}}\\[0.5cm]
  \end{center}
  \small
  Questa tesi presenta la metodologia per la selezione dinamica della
  dimensione del campione negli algoritmi di ottimizzazione per il machine
  learning su larga scala, fondandosi sul lavoro di Byrd, Chin, Nocedal e Wu
  (2012). L'idea centrale è che, invece di fissare a priori la dimensione del
  batch, si possa decidere durante l'algoritmo se e quando aumentarla,
  guidandosi da stime statistiche della varianza del gradiente stocastico. Il
  lavoro offre un'analisi teorica completa della convergenza, il miglioramento
  di alcuni risultati mediante disuguaglianze più strette e
  un'implementazione Python dei quattro algoritmi principali: gradiente
  dinamico, Newton-CG, estensione $L_1$ e BB-CCV, con un'applicazione web
  interattiva per la sperimentazione.
\end{frame}

% SLIDE 3 — Contesto: ottimizzazione su larga scala
% =====================================================================
\begin{frame}{Contesto: ottimizzazione su larga scala}
  \begin{itemize}
    \item ML su larga scala: dataset con $N \approx 10^6$--$10^9$ esempi, perdita media $J(w)$
    \item Gradiente completo $\nabla J(w)$: costo $O(N\,m)$ a iterazione, proibitivo
    \item Standard: metodi stocastici su batch (SGD, metodi a varianza ridotta)
    \item Il batch è un sottoinsieme dei dati usato a ogni passo
    \item Riferimento: Byrd, Chin, Nocedal e Wu (2012)
  \end{itemize}
\end{frame}

% =====================================================================
% SLIDE 4 — Il problema: la dimensione del mini-batch
% =====================================================================
\begin{frame}{Il problema: la dimensione del mini-batch}
  \begin{itemize}
    \item Batch piccolo $\rightarrow$ iterazioni economiche, ma gradiente rumoroso
      \begin{itemize}
        \item il rumore limita la precisione raggiungibile
      \end{itemize}
    \item Batch grande $\rightarrow$ stima accurata, ma costo proporzionale alla dimensione
    \item Nei metodi classici la dimensione è fissa e scelta in modo euristico
    \item Troppo piccolo: non si converge alla precisione desiderata
    \item Troppo grande: risorse sprecate nelle prime iterazioni
  \end{itemize}
\end{frame}

% =====================================================================
% SLIDE 5 — L'idea: campionamento dinamico
% =====================================================================
\begin{frame}{L'idea: campionamento dinamico}
  \begin{itemize}
    \item Si parte da un batch piccolo: progressi rapidi ed economici
    \item Lo si aumenta progressivamente quando serve precisione
    \item La decisione è guidata dalla Condizione di Controllo della Varianza (CCV)
    \item Risorse concentrate dove servono davvero, con adattamento automatico
      \begin{itemize}
        \item alla difficoltà del problema
      \end{itemize}
    \item Nessuna scelta euristica a priori del batch
  \end{itemize}
\end{frame}

% =====================================================================
% SLIDE 6 — La CCV
% =====================================================================
\begin{frame}{La Condizione di Controllo della Varianza (CCV)}
  \begin{itemize}
    \item Finché la varianza campionaria del gradiente stocastico è piccola
          rispetto alla norma del gradiente, il batch è adeguato:
      \[
        \frac{\|\widehat{\mathcal{V}}\|_1}{n_k} \;\le\; \theta^2\,\|\nabla J(w_k)\|_2^2
      \]
    \item Vicino all'ottimo il gradiente tende a zero: il rumore relativo
          cresce e la CCV impone un batch più grande
    \item Il batch cresce a ``scalini'', esattamente quando la soglia viene superata
    \item $\theta$ piccolo $\rightarrow$ batch cresce presto; $\theta$ grande
          $\rightarrow$ batch piccolo più a lungo
  \end{itemize}
\end{frame}

% =====================================================================
% SLIDE 8 — Line search: Armijo e Wolfe
% =====================================================================
\begin{frame}{Line search: Armijo e Wolfe}
  \fonte{Dalla tesi --- Appendice A, condizioni sul passo $\alpha$}

  \textbf{Armijo} (limite superiore su $f$): il passo $\alpha$ è accettato se
  \[
    f(x_k+\alpha\,p) \;\le\; f(x_k) + c_1\,\alpha\,\nabla f(x_k)^T p,
  \]
  con $c_1\in(0,1)$ (tipicamente $10^{-4}$): la retta
  $f(x_k)+c_1\,\alpha\,\nabla f(x_k)^T p$ funge da upper bound; i passi che la
  superano vengono rifiutati e ridotti.

  \vspace{1em}
  \textbf{Curvatura di Wolfe} (limite inferiore su $g'$), con
  $g(\alpha)=f(x_k+\alpha\,p)$ e $g'(\alpha)=\nabla f(x_k+\alpha\,p)^T p$:
  \[
    g'(\alpha) \;\ge\; c_2\,g'(0),
    \qquad c_2\in(c_1,1), \qquad g'(0)=\nabla f(x_k)^T p < 0.
  \]
  La condizione $0<c_1<c_2<1$ garantisce un intervallo non vuoto di passi
  accettabili: Armijo scarta i passi troppo grandi, Wolfe quelli troppo piccoli.
\end{frame}

% =====================================================================
% SLIDE 9 — La CCV (definizione e regola di aggiornamento)
% =====================================================================
\begin{frame}{La Condizione di Controllo della Varianza (CCV)}
  \fonte{Dalla tesi --- Sez.~5.1, Metodo del Gradiente a Campione Dinamico}

  Varianza campionaria del gradiente stimato sul batch $\mathcal{S}_k$:
  \[
    \widehat{\mathcal{V}} \;=\;
    \mathrm{Var}_{i\in\mathcal{S}_k}\!\bigl(\nabla \ell(w_k;i)\bigr)
    \;=\; \frac{1}{n_k-1}\sum_{i\in\mathcal{S}_k}
    \Bigl(\nabla \ell(w_k;i) - \nabla J_{\mathcal{S}_k}(w_k)\Bigr)^2 .
  \]
  La varianza stimata deve essere piccola rispetto al gradiente corrente:
  \[
    \frac{\|\widehat{\mathcal{V}}\|_1}{n_k} \;\le\;
    \theta^2\,\|\nabla J_{\mathcal{S}_k}(w_k)\|_2^2,
    \qquad \theta\in(0,1).
  \]
  Se la CCV è violata, la nuova dimensione del batch è
  \[
    n_k^{\mathrm{new}} \;=\;
    \left\lceil \frac{\|\widehat{\mathcal{V}}\|_1}{\theta^2\,\|\nabla J_{\mathcal{S}_k}(w_k)\|_2^2} \right\rceil .
  \]
\end{frame}

% =====================================================================
% SLIDE 10 — La CCV garantisce una direzione di discesa
% =====================================================================
\begin{frame}{La CCV garantisce una direzione di discesa}
  \fonte{Dalla tesi --- Sez.~5.1, condizione di discesa e Lemma di accuratezza}

  $-g_k$ è di discesa per $J$ se l'errore relativo del gradiente stimato è piccolo:
  \[
    \|e_k\|_2 \;\le\; \theta\,\|g_k\|_2,
    \qquad e_k := g_k - \nabla J(w_k), \qquad \theta\in[0,1).
  \]
  Infatti, ponendo $g_k = \nabla J(w_k) + e_k$ e usando
  Cauchy--Schwarz ($|e_k^T g_k| \le \|e_k\|\,\|g_k\|$):
  \[
    \nabla J(w_k)^T g_k \;\ge\; \|g_k\|^2 - \|e_k\|\,\|g_k\|
    \;=\; \|g_k\|\bigl(\|g_k\| - \|e_k\|\bigr) \;>\; 0
    \qquad\Longrightarrow\qquad -g_k \text{ scende.}
  \]
  Elevando al quadrato, $\|e_k\|_2^2 \le \theta^2\|g_k\|_2^2$; e poiché
  $\mathbb{E}[g_k]=\nabla J(w_k)$:
  \[
    \mathbb{E}\bigl[\|e_k\|_2^2\bigr]
    \;=\; \mathbb{E}\bigl[\|g_k - \mathbb{E}[g_k]\|_2^2\bigr]
    \;=\; \|\mathrm{Var}(g_k)\|_1.
  \]
  Stimando $\|\mathrm{Var}(g_k)\|_1 \approx \|\widehat{\mathcal{V}}\|_1/n_k$,
  la disuguaglianza in attesa diventa esattamente la CCV:
  \[
    \frac{\|\widehat{\mathcal{V}}\|_1}{n_k} \;\le\; \theta^2\,\|g_k\|_2^2 .
  \]
\end{frame}

% SLIDE 15 — Risultati teorici
% =====================================================================
\begin{frame}{Risultati teorici}
  \begin{itemize}
    \item Sotto la CCV la direzione di ricerca è di discesa e il metodo
          converge linearmente (forma deterministica e in aspettativa)
    \item Complessità totale: $O(mL/(\lambda\varepsilon))$ --- competitiva con SGD
    \item Su problemi mal condizionati il termine $\kappa^2$ di SGD diventa $\kappa$
    \item Fattore di contrazione migliorato: da
          $1-\frac{(1-\theta)\lambda}{2L}$ a
          $1-\frac{(1-\theta)\lambda}{L}$ usando la disuguaglianza di
          Polyak--Lojasiewicz in forma forte
  \end{itemize}
\end{frame}

\begin{frame}{Riuso del mini-batch (Sez.~6.7)}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Variante: riusare lo stesso campione per $M$ iterazioni consecutive
        \item Si riusa la lista degli indici, NON il gradiente
          \begin{itemize}
            \item ricalcolato a ogni iterazione al punto corrente
          \end{itemize}
        \item La CCV fa da segnale di esaurimento: se non è soddisfatta,
              $n_k$ cresce e si ricampiona
        \item $M=\infty$: si tiene il campione finché la CCV vale
        \item L'effetto dipende dal metodo e dal condizionamento del problema
      \end{itemize}
    \end{column}
